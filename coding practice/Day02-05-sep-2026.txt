Ansible - Install & Enable Nginx

- name:Install Nginx
 hosts: all
 become:yes

 task:
  - name: Install nginx
    yum:
     name: nginx
     state: present


  - name: Enable nginx
    service:
    name: nginx
    state: started
    enabled: yes
