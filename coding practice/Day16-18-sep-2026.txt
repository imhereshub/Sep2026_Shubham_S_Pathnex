🔹 Ansible — Create and Start EC2 Instances
- name: create EC2 instance
  hosts: localhost
  become: yes
  tasks:
    - name: Launch EC2 instance in us-east-1
      ec2_instance:
        name: pathnex-ec2-us-east-1
        key_name: pathne-key
        region: us-east-1
        instance_type: t2.micro
        image: ami-123456789
        wait: yes
    - name: Launch EC2 instance in us-west-1
      ec2_instance:
        name: pathnex-ec2-us-west-1
        key_name: pathnex-key
        region: us-west-1
        image: ami-123456788
        instance_type: t2.micro
        wait: yes

🔹 Jenkinsfile — Deploy to Multi-Cluster Environments
pipeline {
   agent any 
   stages {
      stage('build') {
       step {
         echo 'Building Docker image..'
         docker.build('pathnex-web-app')
        }
      } 
      stage('test') {
       step {
        echo 'Running test..'
      }
    }
      stage('Deploy to kubernets') {
       step {
         script {
           if (env.CLUSTER == 'us-east') {
               sh 'kubectl --context=us-east apply -f deployment.yaml
           } else {
               sh 'kubectl --context=us-west apply -f deployment.yaml
           }
         }
       }
    }
  }
}
🔹 GitLab CI/CD — Multi-Region Deployment with Kubernetes
stages:
 - build
 - Push
 - deploy

 stage: build
 script:
   - docker build -t pathnex-web-app .

   
deploy:
  stage: deploy
  script:
    - |
     if [ "$CI_COMMIT_REF_NAME" == "main" ]; then
       kubectl -- context==us-east apply -f deployment.yaml
     else 
       kubectl -- context==us-west apply -f kubernetes/dev-deployment.yaml
     fi
